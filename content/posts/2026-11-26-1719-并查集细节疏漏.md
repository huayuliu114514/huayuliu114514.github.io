并查集，思路上来说其实算是比较简单的数据结构，但实现起来有很多坑和细节
1、初始化g[i]=i——牢记“根是自己”“根是自己”“根是自己”
2、unite和find函数是板子，本身要记住
void unite(vector<int>&g,int a,int b){
int ra=find(g,a),rb=find(g,b);
if(ra!=rb)g[ra]=rb;
}
int find(vector<int>&us,int x){
if(us[x]==x)return x;
return us[x]=find(us,us[x]);
}
